// -------------------- Pins --------------------
#define DHTPIN 4
#define DHTTYPE DHT11

#define MQ2_PIN 34
#define FLAME_PIN 27
#define BUZZER_PIN 2  // <-- Buzzer Pin (GPIO 25)

// -------------------- Objects --------------------
DHT dht(DHTPIN, DHTTYPE);
FirebaseData fbdo;
FirebaseConfig config;
FirebaseAuth auth;

// -------------------- Variables --------------------
float temperature;
int mq2Value;
int flameValue;
String systemStatus;

// -------------------- Setup --------------------
void setup()
{
  Serial.begin(115200);

  dht.begin();
  pinMode(FLAME_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);   // <-- Setting buzzer pin as Output
  digitalWrite(BUZZER_PIN, LOW);  // Start with buzzer OFF

  Serial.println("================================");
  Serial.println("SMART FIRE DETECTION SYSTEM");
  Serial.println("================================");

  // -------- Connect to WiFi --------
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("Connecting to Wi-Fi");
  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConnected to Wi-Fi successfully!");
  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());

  // -------- Configure Firebase Settings --------
  config.database_url = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;

  Firebase.reconnectWiFi(true); // Auto-reconnect to Wi-Fi if dropped
  Firebase.begin(&config, &auth);
}

// -------------------- Loop --------------------
void loop()
{
  // Read Sensors
  temperature = dht.readTemperature();
  mq2Value = analogRead(MQ2_PIN);
  flameValue = digitalRead(FLAME_PIN);

  // Handle sensor read failure gracefully so it doesn't pollute your dataset
  if (isnan(temperature)) {
    Serial.println("Error: Failed to read from DHT sensor! Skipping cycle.");
    delay(2000);
    return; 
  }

  // Default Status
  systemStatus = "SAFE";

  // -------- Corrected Classification Logic (ML Ready!) --------
  if (mq2Value < 2200 && flameValue == 1 && temperature < 40)
  {
    systemStatus = "SAFE";
  }
  else if (flameValue == 0 && temperature >= 45) 
  {
    // Aag bhi lagi hui hai aur temperature bhi high hai -> Critical
    systemStatus = "CRITICAL FIRE";
  }
  else if (flameValue == 0 && temperature < 45)
  {
    // Sirf Flame detect hui hai, abhi temperature kam hai -> Fire Alert
    systemStatus = "FIRE ALERT";
  }
  else if (mq2Value > 2200 && flameValue == 1 && temperature < 40)
  {
    systemStatus = "GAS LEAK WARNING";
  }
  else if (mq2Value > 2200 && flameValue == 1 && temperature >= 40)
  {
    systemStatus = "SMOKE WARNING";
  }

  // -------- Display Local Results --------
  Serial.println("\n================================");
  Serial.print("Temperature : ");
  Serial.print(temperature);
  Serial.println(" C");

  Serial.print("MQ2 Value   : ");
  Serial.println(mq2Value);

  Serial.print("Flame Value : ");
  Serial.println(flameValue);

  Serial.print("STATUS      : ");
  Serial.println(systemStatus);

  // -------- Dataset Format for Serial Monitor Logger --------
  Serial.print("CSV_DATA: ");
  Serial.print(temperature);
  Serial.print(",");
  Serial.print(mq2Value);
  Serial.print(",");
  Serial.print(flameValue);
  Serial.print(",");
  Serial.println(systemStatus);

  // -------- Update Firebase Realtime Database --------
  if (Firebase.ready())
  {
    FirebaseJson json;
    json.set("temperature", temperature);
    json.set("mq2", mq2Value);
    json.set("flame", flameValue);
    json.set("status", systemStatus);

    // Update the nodes under /fireSystem path seamlessly for Flutter
    if (Firebase.RTDB.updateNode(&fbdo, "/fireSystem", &json))
    {
      Serial.println("Cloud Sync: Success ✅");
    }
    else
    {
      Serial.print("Cloud Sync: Failed ❌ Reason: ");
      Serial.println(fbdo.errorReason());
    }
  }
  else
  {
    Serial.println("Cloud Sync: Firebase connection not ready...");
  }

  // -------- New Buzzer Control Patterns --------
  if (systemStatus == "FIRE ALERT" || systemStatus == "CRITICAL FIRE")
  {
    // Constant Alarm for Fire (Continuous loud beep for 4 seconds)
    digitalWrite(BUZZER_PIN, HIGH);
    delay(4000); 
  }
  else if (systemStatus == "GAS LEAK WARNING" || systemStatus == "SMOKE WARNING")
  {
    // Intermittent Beeping Alarm for Gas (Beep... delay... Beep...)
    for (int i = 0; i < 4; i++) { 
      digitalWrite(BUZZER_PIN, HIGH);
      delay(400);
      digitalWrite(BUZZER_PIN, LOW);
      delay(600);
    }
  }
  else
  {
    // System is SAFE, turn off buzzer completely
    digitalWrite(BUZZER_PIN, LOW);
    delay(4000); 
  }
}
